for epoca in range(epochs):
    parametri.train()
    optimz.zero_grad()  #azzera i gradienti calcolati nell'epoca precedente

    previsioneTRAIN = parametri(xTRAIN, archiIDX) #passa la sequenza di training nel modello e ottieni la previsione su settimana (come da architettura)

    targetTRAIN = yTRAIN[-1] #confronto target con valore effettivo dell'ultima settimana

    #TROPPI VALORI A 0, AGGIUSTO LOSS DI MODO CHE PREVISIONE SBAGLIATA SU 1 SIA MOLTO DISINCENTIVATA
    pesi = torch.where(targetTRAIN == 1, posWG, torch.tensor(1.0))
    lossTRAIN = nn.functional.binary_cross_entropy(previsioneTRAIN, targetTRAIN, reduction='none') #calcolo loss per ogni comune
    lossTOT = (lossTRAIN * pesi).mean()

    lossTOT.backward()      #calcola gradienti (quanto ogni peso ha contribuito a errore)
    optimz.step()    #aggiorna pesi della rete in base a gradienti

    if epoca % 10 == 0:
        print(f"Epoca {epoca} - Loss training: {lossTOT.item():.4f}")
